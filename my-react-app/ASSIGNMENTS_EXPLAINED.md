## assign.c

Purpose:
- Demonstrates **time complexity** by timing constant work O(1), a single loop O(n), and a nested loop O(n²) for different input sizes.

Annotated file:
```c
#include <stdio.h>   // Gives access to printf() for printing output.
#include <time.h>    // Gives access to clock_t, clock(), and CLOCKS_PER_SEC.

int main() {         // Program entry point.
    int n;           // Will hold the current input size we’re testing.
    clock_t start, end; // CPU clock tick counts for start and end times.
    double t;        // Stores elapsed time in seconds.

    int inputs[] = {1000, 5000, 10000}; // Test sizes for n.

    for (int k = 0; k < 3; k++) {      // Loop over the 3 input sizes.
        n = inputs[k];                 // Pick one size.
        printf("\nInput size = %d\n", n); // Print header for this test.

        /* Constant Time O(1) */
        start = clock();               // Start timing.
        int a = 5, b = 10, c;          // Declare 3 ints; a and b are constants.
        c = a + b;                     // One addition: constant amount of work.
        end = clock();                 // Stop timing.
        t = (double)(end - start) / CLOCKS_PER_SEC; // Convert ticks to seconds.
        printf("Constant time O(1): %f seconds\n", t); // Print measured time.

        /* Linear Time O(n) */
        start = clock();               // Start timing the linear loop.
        int sum = 0;                   // Accumulator so the loop “does work”.
        for (int i = 0; i < n; i++) {  // Runs n times ⇒ O(n).
            sum = sum + i;             // Constant work each iteration.
        }
        end = clock();
        t = (double)(end - start) / CLOCKS_PER_SEC;
        printf("Linear time O(n): %f seconds\n", t);

        /* Quadratic Time O(n^2) */
        start = clock();               // Start timing the nested loops.
        int count = 0;                 // Counter to do some work in the loop.
        for (int i = 0; i < n; i++) {  // Outer loop runs n times.
            for (int j = 0; j < n; j++) { // Inner loop runs n times per i.
                count++;               // Total increments ≈ n*n ⇒ O(n²).
            }
        }
        end = clock();
        t = (double)(end - start) / CLOCKS_PER_SEC;
        printf("Quadratic time O(n^2): %f seconds\n", t);
    }

    return 0;                          // Return 0 means “success”.
}
```

Extra notes:
- `clock()` measures CPU time ticks; for very fast blocks like O(1), the printed time may be 0.000000 depending on resolution.

## assign2.c

Purpose:
- Demonstrates **space complexity** ideas: O(1) memory (a few variables), O(n) memory (an array), and O(n²) memory (a matrix).

Annotated file:
```c
#include <stdio.h>           // Gives access to printf().

int main() {                 // Program entry point.
    int n = 5;               // “n” is 5 here; used for loops and array sizes.

    int a = 10, b = 20, c;   // 3 integers take constant memory ⇒ O(1) space.
    c = a + b;               // Constant computation.
    printf("Constant Space O(1)\n"); // Print a label.

    int arr[5];              // Array of 5 ints ⇒ O(n) space in general.
    for (int i = 0; i < n; i++) { // Fill array.
        arr[i] = i;          // Store i at position i.
    }
    printf("Linear Space O(n)\n"); // Print a label.

    int matrix[5][5];        // 2D array 5x5 ⇒ O(n²) space in general.
    for (int i = 0; i < n; i++) {      // Outer loop over rows.
        for (int j = 0; j < n; j++) {  // Inner loop over columns.
            matrix[i][j] = i + j;      // Store something in each cell.
        }
    }
    printf("Quadratic Space O(n^2)\n"); // Print a label.

    return 0;               // Success.
}
```

Important detail:
- In this exact program `n` is hard-coded to 5, so the memory is actually fixed. The “O(n)” and “O(n²)” labels are teaching the *concept*.

## assign2.py

Purpose:
- An “Activity Log Analyzer” that:
  1) Computes total time per user (using `reduce()`),
  2) Finds the top K most active users,
  3) Extracts the set of unique actions.

Annotated file:
```py
"""                       # Module docstring (a description of the script).
Assignment 7: Activity Log Analyzer
"""

from collections import defaultdict # defaultdict lets you auto-initialize missing keys.
from functools import reduce        # reduce() folds a list into a single value.
from typing import Any              # Any is used in the type hints below.

# ── Sample Data ────────────────────────────────────────────────────────────────

logs: list[dict[str, Any]] = [      # logs is a list of dicts; each dict is one record.
    {"user": "CS101", "action": "YouTube",     "duration": 45.0},
    {"user": "CS102", "action": "Instagram",   "duration": 30.0},
    {"user": "CS101", "action": "Google Docs", "duration": 60.0},
    {"user": "CS103", "action": "YouTube",     "duration": 90.0},
    {"user": "CS102", "action": "Zoom",        "duration": 50.0},
    {"user": "CS101", "action": "Zoom",        "duration": 40.0},
    {"user": "CS104", "action": "Netflix",     "duration": 120.0},
    {"user": "CS103", "action": "Google Docs", "duration": 35.0},
    {"user": "CS104", "action": "YouTube",     "duration": 80.0},
    {"user": "CS102", "action": "Netflix",     "duration": 70.0},
]


# ── 1. Total time per user using reduce() ──────────────────────────────────────

def total_time_per_user(logs: list[dict]) -> dict[str, float]: # Function returns {user: total_duration}.
    def add_duration(acc: defaultdict, record: dict) -> defaultdict: # Inner function used by reduce.
        acc[record["user"]] += record["duration"]  # Add this record’s duration to the user’s total.
        return acc                                  # Return the updated accumulator.

    return dict(reduce(add_duration, logs, defaultdict(float))) # Fold all records into one totals map.


# ── 2. Top K most active users ─────────────────────────────────────────────────

def most_active_users(logs: list[dict], k: int) -> list[str]:  # Return the top-k user IDs.
    totals = total_time_per_user(logs)                         # Compute totals first.
    ranked = sorted(totals, key=lambda user: totals[user], reverse=True) # Sort by total duration.
    return ranked[:k]                                          # Slice the first k users.


# ── 3. Unique actions using a set ──────────────────────────────────────────────

def unique_actions(logs: list[dict]) -> set[str]:              # Return distinct action names.
    return {record["action"] for record in logs}              # Set comprehension removes duplicates.


# ── Run & Print Results ────────────────────────────────────────────────────────

print("=== Total Time Per User ===")      # Section header.
totals = total_time_per_user(logs)         # Compute totals.
for user, mins in totals.items():          # Loop through each user and duration.
    print(f"  {user}: {mins} min")         # Print using an f-string.

print("\n=== Top 3 Most Active Users ===") # New section header (\n adds a blank line).
top3 = most_active_users(logs, k=3)        # Compute top 3.
for i, user in enumerate(top3, start=1):   # enumerate gives (rank_index, user).
    print(f"  #{i} {user} — {totals[user]} min") # Print ranking and time.

print("\n=== Unique Actions ===")         # New section.
actions = unique_actions(logs)             # Compute unique actions.
print(f"  {actions}")                     # Print the set.

print("\n=== Complexity Analysis ===")    # New section.
print("  total_time_per_user -> Time: O(n),            Space: O(u)")
print("  most_active_users   -> Time: O(n + u·log u),  Space: O(u)")
print("  unique_actions      -> Time: O(n),            Space: O(a)")
print("\n  n=log entries, u=unique users, a=unique actions") # Explain symbols.
```

Why these complexities:
- `total_time_per_user`: one pass through `logs` ⇒ O(n) time; stores totals per user ⇒ O(u) space.
- `most_active_users`: totals is O(n) + sort `u` users ⇒ O(u log u).
- `unique_actions`: one pass through logs building a set ⇒ O(n) time, O(a) space.

## assing.java

Purpose:
- Models students with courses + scores, then uses Java Streams to:
  1) Get top N students by average score,
  2) Compute average score per course,
  3) Collect all unique courses.

Annotated file:
```java
import java.util.ArrayList;          // For resizable lists.
import java.util.Arrays;             // For Arrays.asList().
import java.util.Comparator;         // For sorting (Comparator).
import java.util.HashMap;            // For maps (course -> score).
import java.util.HashSet;            // For sets (unique courses).
import java.util.List;               // List interface.
import java.util.Map;                // Map interface.
import java.util.Set;                // Set interface.
import java.util.stream.Collectors;  // Stream collectors (groupingBy, averaging, etc.).

class Student {                      // Defines a Student type (package-private).
    private int id;                  // Student ID.
    private String name;             // Student name.
    private List<String> courses;    // Course names the student is enrolled in.
    private Map<String, Integer> scores; // Scores per course.

    public Student(int id, String name, List<String> courses, Map<String, Integer> scores) {
        this.id = id;                // Save id.
        this.name = name;            // Save name.
        this.courses = new ArrayList<>(courses); // Copy list so caller’s list can’t mutate ours.
        this.scores = new HashMap<>(scores);     // Copy map for the same reason.
    }

    public int getId() {             // Getter for id.
        return id;
    }

    public String getName() {        // Getter for name.
        return name;
    }

    public List<String> getCourses() { // Getter for courses (note: returns internal list).
        return courses;
    }

    public Map<String, Integer> getScores() { // Getter for scores (note: returns internal map).
        return scores;
    }

    public double getAverageScore() { // Compute average score across the student’s courses.
        if (courses.size() == 0) {    // Guard against divide-by-zero and empty streams.
            return 0;
        }

        return courses.stream()       // Stream over course names.
                .mapToInt(course -> scores.getOrDefault(course, 0)) // Convert course -> score (0 if missing).
                .average()            // Compute average as OptionalDouble.
                .orElse(0);           // If no values, return 0.
    }
}

public class assing {                // Main class (name is lowercase; Java usually uses Assing).

    public static List<Student> getTopNStudents(List<Student> students, int n) {
        return students.stream()      // Stream the list.
                .sorted(Comparator.comparingDouble(Student::getAverageScore).reversed()) // Sort by avg desc.
                .limit(n)             // Keep first n.
                .collect(Collectors.toCollection(ArrayList::new)); // Collect into a new ArrayList.
    }

    public static Map<String, Double> getAverageScorePerCourse(List<Student> students) {
        return students.stream()      // Stream students.
                .flatMap(student -> student.getCourses().stream()  // For each student, stream their courses...
                        .map(course -> Map.entry(course, student.getScores().getOrDefault(course, 0))))
                        // ...and turn each (student, course) into an entry (course -> score).
                .collect(Collectors.groupingBy(
                        Map.Entry::getKey,      // Group by course name.
                        HashMap::new,           // Use a HashMap to store groups.
                        Collectors.averagingInt(Map.Entry::getValue) // Average the scores in each group.
                ));
    }

    public static Set<String> getAllUniqueCourses(List<Student> students) {
        return students.stream()      // Stream students.
                .flatMap(student -> student.getCourses().stream()) // Flatten all course lists into one stream.
                .collect(Collectors.toCollection(HashSet::new));   // Collect distinct courses into a HashSet.
    }

    public static void main(String[] args) { // Entry point.
        List<Student> students = new ArrayList<>(); // Create student list.

        students.add(new Student(1, "Rahul",
                new ArrayList<>(Arrays.asList("Math", "Java", "DBMS")),
                new HashMap<>(Map.of("Math", 85, "Java", 90, "DBMS", 80))));
                // Student 1 with 3 courses and scores.

        students.add(new Student(2, "Priya",
                new ArrayList<>(Arrays.asList("Math", "Java", "OS")),
                new HashMap<>(Map.of("Math", 92, "Java", 88, "OS", 84))));

        students.add(new Student(3, "Arun",
                new ArrayList<>(Arrays.asList("Java", "DBMS", "OS")),
                new HashMap<>(Map.of("Java", 75, "DBMS", 81, "OS", 79))));

        students.add(new Student(4, "Sneha",
                new ArrayList<>(Arrays.asList("Math", "DBMS", "CN")),
                new HashMap<>(Map.of("Math", 89, "DBMS", 91, "CN", 87))));

        students.add(new Student(5, "Kiran",
                new ArrayList<>(Arrays.asList("Java", "CN", "OS")),
                new HashMap<>(Map.of("Java", 70, "CN", 73, "OS", 68))));

        System.out.println("Top 3 students based on average score:"); // Header.
        List<Student> topStudents = getTopNStudents(students, 3);      // Compute top 3.
        for (Student student : topStudents) {                          // Loop through them.
            System.out.println(student.getName() + " - Average: " + student.getAverageScore());
        }

        System.out.println();                                          // Blank line.
        System.out.println("Average score per course:");               // Header.
        Map<String, Double> avgMap = getAverageScorePerCourse(students); // Compute averages.
        for (Map.Entry<String, Double> entry : avgMap.entrySet()) {     // Loop entries.
            System.out.println(entry.getKey() + " : " + entry.getValue());
        }

        System.out.println();                                          // Blank line.
        System.out.println("All unique courses:");                     // Header.
        Set<String> courses = getAllUniqueCourses(students);           // Compute unique courses.
        for (String course : courses) {                                // Print each one.
            System.out.println(course);
        }

        System.out.println();
        System.out.println("Complexity Analysis:");                    // Header.
        System.out.println("1. Time complexity of computing course averages is O(S * C)");
        System.out.println("   because we go through each student's course list once.");
        System.out.println("2. Complexity of sorting top N students is O(S * C + S log S)");
        System.out.println("   because average is computed and students are sorted.");
    }
}
```

Notes:
- The getters return the internal `List` and `Map`. If external code modifies them, it can change the student. A common improvement is returning defensive copies (not required for the assignment, just a design note).

```tsx
import { useState } from "react";   // useState stores component state.

function TodoApp() {                // Defines a React component.
  const [todos, setTodos] = useState([]); // todos array; starts empty. (In strict TS you’d type this.)
  const [task, setTask] = useState("");   // current input text.

  const addTodo = () => {            // Function called when clicking “Add”.
    setTodos([...todos, task]);      // Create a NEW array with previous todos + new task.
    setTask("");                    // Clear the input.
  };

  return (                           // Return JSX UI.
    <div>
      <h3>Todo List</h3>             {/* Heading */}

      <input
        type="text"                 // Text input.
        value={task}                 // Controlled input: value comes from state.
        onChange={(e) => setTask(e.target.value)} // Update state when user types.
      />

      <button onClick={addTodo}>Add</button> // Button triggers addTodo.

      <ul>
        {todos.map((t, index) => (   // Render one <li> per todo.
          <li key={index}>{t}</li>   // key helps React track list items.
        ))}
      </ul>
    </div>
  );
}

export default TodoApp;              // Export as default so other files can import it.
```

- As written, clicking Add with an empty `task` will add an empty item; you could prevent that with a small validation check.

Purpose:
- Custom hook that holds all state + derived data for the dashboard:
  - Students stored in a `Map` keyed by ID,
  - Unique courses derived using a `Set`,
  - Sorting/filtering logic,
  - Add/remove student operations.

```ts
import { useState } from "react";              // React state.
import type { NewStudentForm, Student } from "./types"; // Types.

const initialStudents = new Map<number, Student>([ // Initial data as a Map (id -> student).
  [
    101,
    {
      id: 101,
      name: "Ananya",
      enrolledCourses: new Set(["React", "DBMS", "CN"]),
      gpa: 8.9,
    },
  ],
  [
    102,
    {
      id: 102,
      name: "Rohit",
      enrolledCourses: new Set(["OS", "React", "Java"]),
      gpa: 8.1,
    },
  ],
  [
    103,
    {
      id: 103,
      name: "Meera",
      enrolledCourses: new Set(["Java", "DBMS"]),
      gpa: 9.2,
    },
  ],
  [
    104,
    {
      id: 104,
      name: "Karthik",
      enrolledCourses: new Set(["CN", "OS"]),
      gpa: 7.8,
    },
  ],
]);

const emptyForm: NewStudentForm = {   // Resets the controlled form inputs.
  id: "",
  name: "",
  courses: "",
  gpa: "",
};

export function useEnrollmentDashboard() { // Custom hook used by the dashboard component.
  const [studentsById, setStudentsById] = useState<Map<number, Student>>( // State: Map of students.
    () => new Map(initialStudents), // Lazy init: create a fresh Map so state isn’t shared.
  );
  const [selectedCourse, setSelectedCourse] = useState("All Courses"); // State: dropdown filter.
  const [form, setForm] = useState<NewStudentForm>(emptyForm);          // State: form input values.

  const students = [...studentsById.values()]; // Convert Map values into an array for sorting/filtering.

  const uniqueCourses = students.reduce((courseSet, student) => { // Build a Set of all courses.
    [...student.enrolledCourses].forEach((course) => courseSet.add(course)); // Add each course.
    return courseSet; // Return accumulator for next reduce step.
  }, new Set<string>()); // Start with an empty Set.

  const sortedStudents = [...students].sort((first, second) => second.gpa - first.gpa); // Sort by GPA desc.

  const filteredStudents = sortedStudents.filter((student) => { // Apply course filter.
    if (selectedCourse === "All Courses") {
      return true; // No filtering; include everyone.
    }

    return student.enrolledCourses.has(selectedCourse); // Set membership test (avg O(1)).
  });

  function updateForm(field: keyof NewStudentForm, value: string) { // Update one form field.
    setForm((current) => ({
      ...current, // Keep other fields the same.
      [field]: value, // Update the chosen field.
    }));
  }

  function addStudent() { // Read form values, validate, then insert student into Map.
    const id = Number(form.id.trim());    // Convert id string -> number.
    const name = form.name.trim();        // Trim whitespace.
    const gpa = Number(form.gpa.trim());  // Convert gpa string -> number.
    const courseList = form.courses       // Turn comma-separated courses into a list.
      .split(",")
      .map((course) => course.trim())
      .filter(Boolean);                   // Remove empty strings.

    if (!id || !name || Number.isNaN(gpa) || courseList.length === 0) {
      return; // If invalid input, do nothing.
    }

    const newStudent: Student = {         // Build Student object.
      id,
      name,
      enrolledCourses: new Set([...courseList]), // Store as Set for fast has().
      gpa,
    };

    setStudentsById((current) => {        // Functional update to avoid stale state.
      const next = new Map(current);      // Copy Map (immutability for React state).
      next.set(id, newStudent);           // Add/replace by id.
      return next;                        // New Map becomes next state.
    });

    setForm({ ...emptyForm });            // Reset form.
  }

  function removeStudent(id: number) {    // Remove a student by id.
    setStudentsById((current) => {
      const next = new Map(current);      // Copy Map.
      next.delete(id);                    // Delete by key.
      return next;
    });
  }

  return {                                // Expose state + actions to UI components.
    form,
    students: filteredStudents,
    totalStudents: students.length,
    uniqueCourses: [...uniqueCourses].sort(), // Convert Set -> array for rendering, sort alphabetically.
    selectedCourse,
    setSelectedCourse,
    updateForm,
    addStudent,
    removeStudent,
  };
}
```
